ENG_settlement_region_lost = {
	OR = {
		AND = {
			check_variable = { ENG_referendum_target = 1 }
			OR = {
				country_exists = SCO
				has_country_flag = ENG_scottish_integrated
				NOT = { ENG_holds_scotland = yes }
			}
		}
		AND = {
			check_variable = { ENG_referendum_target = 2 }
			OR = {
				country_exists = NIR
				has_country_flag = ENG_north_irish_integrated
				NOT = { ENG_holds_north_ireland = yes }
			}
		}
		AND = {
			check_variable = { ENG_referendum_target = 3 }
			OR = {
				country_exists = WAS
				has_country_flag = ENG_welsh_integrated
				NOT = { ENG_holds_wales = yes }
			}
		}
	}
}

# Target scope with ROOT = ENG. influence_higher_50 plus the La Resistance network branch the foreign-ops focuses use.
ENG_uq_high_influence = {
	if = {
		limit = { has_dlc = "La Resistance" }
		ENG = { network_strength = { target = PREV value > 50 } }
	}
	else = {
		check_variable = { influence_array^0 = ENG }
		check_variable = { influence_array_val^0 > 49.999 }
	}
}

# Target scope with ROOT = ENG. influence_higher_30 plus the La Resistance network branch the foreign-ops focuses use.
ENG_uq_medium_influence = {
	if = {
		limit = { has_dlc = "La Resistance" }
		ENG = { network_strength = { target = PREV value > 30 } }
	}
	else = {
		any_of = {
			array = influence_array
			value = v
			index = i
			var:v = { tag = ENG }
			check_variable = { influence_array_val^i > 29.999 }
		}
	}
}

# FROM = the ladder target. True while every state it holds a core on is quiet under British rule.
ENG_uq_states_pacified = {
	NOT = {
		any_state = {
			is_core_of = FROM
			OR = {
				compliance < 70
				resistance > 20
			}
		}
	}
}

# Target scope with ROOT = ENG. Every state it holds a core on is under British rule.
ENG_uq_all_cores_held = {
	any_state = { is_core_of = PREV }
	NOT = {
		any_state = {
			is_core_of = PREV
			NOT = {
				OR = {
					is_controlled_by = ROOT
					controller = { is_subject_of = ROOT }
				}
			}
		}
	}
}
